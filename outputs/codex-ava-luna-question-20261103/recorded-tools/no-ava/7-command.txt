/bin/zsh -lc "python3 - <<'PY'
import csv
from collections import defaultdict
with open('inputs/sales.csv',newline='') as f: rows=list(csv.DictReader(f))
# pay-cohort: paid line amount less refunds associated with the cohort (refund dates cross-month included)
base=[r for r in rows if r['order_kind']=='customer_sale' and r['paid_at'].startswith(('2026-08','2026-09'))]
# aggregate values via paid month, with refund amount attributed to original sale month
for c in ['2026-08','2026-09']:
 rr=[r for r in base if r['paid_at'].startswith(c)]
 paid=sum(float(r['line_paid_amount'] or 0) for r in rr)
 refund=sum(float(r['refund_amount'] or 0) for r in rr)
 print(c,'rows',len(rr),'paid',paid,'refund',refund,'net',paid-refund)
 for dim in ['business_model','channel','category','selling_region','product_name']:
  d=defaultdict(float)
  for r in rr: d[r[dim]]+=float(r['line_paid_amount'] or 0)-float(r['refund_amount'] or 0)
  print(dim, sorted(d.items(),key=lambda x:-x[1])[:20])
# refunds by refunded_at month
for c in ['2026-08','2026-09']:
 rr=[r for r in rows if r['order_kind']=='customer_sale' and r['refunded_at'].startswith(c)]
 print('refund calendar',c,sum(float(r['refund_amount'] or 0) for r in rr),len(rr))
PY"
