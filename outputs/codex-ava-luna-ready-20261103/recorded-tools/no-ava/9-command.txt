from pathlib import Path
import csv
from decimal import Decimal
import matplotlib.pyplot as plt
import numpy as np
rows=list(csv.DictReader(open('inputs/sales.csv',newline='',encoding='utf-8-sig')))
def net(r): return Decimal(r['line_paid_amount'] or '0')-Decimal(r['refund_amount'] or '0')
months=['2026-08','2026-09']; vals=[]
for m in months:
 vals.append(sum(net(r) for r in rows if r['paid_at'].startswith(m) and r['order_kind']=='customer_sale'))
delta=vals[1]-vals[0]; pct=delta/vals[0]*100
fields=['business_model','selling_region','category','channel','selling_city','product_name']
breakdowns={}
for f in fields:
 d={m:{} for m in months}
 for r in rows:
  m=r['paid_at'][:7]
  if m in d and r['order_kind']=='customer_sale': d[m][r[f]]=d[m].get(r[f],Decimal(0))+net(r)
 breakdowns[f]=[(k,d[months[1]].get(k,0)-d[months[0]].get(k,0)) for k in set(d[months[0]])|set(d[months[1]])]
print('TOTALS',vals,'delta',delta,'pct',pct)
for f in fields:
 print('\\n',f,sorted(breakdowns[f],key=lambda x:-x[1]))
# Main driver chart
labels=['Self operated','Consignment','North China','South China','East China','Furniture','Digital','Online','Stores']
deltas=[]
for f,k in [('business_model','self_operated'),('business_model','consignment'),('selling_region','华北'),('selling_region','华南'),('selling_region','华东'),('category','家具'),('category','数码'),('channel','线上商城'),('channel','线下门店')]:
 d=dict(breakdowns[f]); deltas.append(float(d.get(k,0))/1e6)
plt.rcParams['font.sans-serif']=['Arial Unicode MS','Noto Sans CJK SC','DejaVu Sans']
plt.rcParams['axes.unicode_minus']=False

axs[0].bar(['Aug','Sep'],[float(v)/1e6 for v in vals],color=['#9db9b4','#2f7d70'],width=.58)
axs[0].set_ylabel('Net sales (CNY million)'); axs[0].set_title('Customer sales by payment month')
for i,v in enumerate(vals): axs[0].text(i,float(v)/1e6+.1,f'{float(v)/1e6:.2f}',ha='center',fontweight='bold')
axs[0].text(.5, max(float(v)/1e6 for v in vals)+.65,f'{float(delta)/1e6:+.2f}m ({float(pct):+.1f}%)',ha='center',color='#2f7d70',fontweight='bold')
axs[0].set_ylim(0,float(vals[1])/1e6+1.05)
# components in grouped dimensions (each is an independent view; do not sum groups)
lab=['Self operated','Consignment','North China','South China','East China','Furniture','Digital','Online','Stores']
y=np.arange(len(lab)); c=['#2f7d70' if v>=0 else '#cc715b' for v in deltas]
axs[1].barh(y,deltas,color=c)
axs[1].set_yticks(y,lab); axs[1].invert_yaxis(); axs[1].axvline(0,color='#555',lw=.8)
axs[1].set_xlabel('Change vs Aug (CNY million)'); axs[1].set_title('Where the change came from')
for i,v in enumerate(deltas): axs[1].text(v+(0.04 if v>=0 else -0.04),i,f'{v:+.2f}',va='center',ha='left' if v>=0 else 'right',fontsize=9)
fig.suptitle('September 2026 net sales rose, but mix and status shifted',fontsize=14,fontweight='bold')
fig.text(.01,.01,'Metric = line_paid_amount − refund_amount; customer_sale only; grouped by paid_at. Breakdown groups are independent views.',fontsize=8,color='#555')
fig.tight_layout(rect=[0,.04,1,.93])
