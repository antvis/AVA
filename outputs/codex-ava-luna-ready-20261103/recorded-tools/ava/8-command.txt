/bin/zsh -lc "cat > output/monthly.sql <<'SQL'
SELECT
  strftime(CAST(completed_at AS DATE), '%Y-%m') AS month,
  SUM(ROUND(CAST(line_paid_amount AS DECIMAL(18,2))*CAST(commission_rate AS DECIMAL(6,4)),2)-ROUND(COALESCE(CAST(refund_amount AS DECIMAL(18,2)),0)*CAST(commission_rate AS DECIMAL(6,4)),2)) AS net_sales,
  COUNT(DISTINCT order_id) AS orders
FROM data
WHERE order_kind='customer_sale' AND status='completed'
  AND CAST(completed_at AS DATE) >= DATE '2026-08-01'
  AND CAST(completed_at AS DATE) < DATE '2026-10-01'
GROUP BY 1 ORDER BY 1;
SQL
ava query ds_sales_4e138c7b982b --dsl @output/monthly.sql > output/monthly.json"
