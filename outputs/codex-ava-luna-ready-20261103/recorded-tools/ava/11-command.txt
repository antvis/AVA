/bin/zsh -lc "cat > output/geo.sql <<'SQL'
WITH base AS (
 SELECT *,
  (ROUND(CAST(line_paid_amount AS DECIMAL(18,2))*CAST(commission_rate AS DECIMAL(6,4)),2)-ROUND(COALESCE(CAST(refund_amount AS DECIMAL(18,2)),0)*CAST(commission_rate AS DECIMAL(6,4)),2)) AS net_line,
  strftime(CAST(completed_at AS DATE), '%Y-%m') AS month
 FROM data
 WHERE order_kind='customer_sale' AND status='completed'
  AND CAST(completed_at AS DATE) >= DATE '2026-08-01'
  AND CAST(completed_at AS DATE) < DATE '2026-10-01'
), g AS (
 SELECT selling_region, SUM(CASE WHEN month='2026-08' THEN net_line ELSE 0 END) AS aug,
        SUM(CASE WHEN month='2026-09' THEN net_line ELSE 0 END) AS sep
 FROM base GROUP BY selling_region
)
SELECT selling_region, aug, sep, sep-aug AS change FROM g ORDER BY ABS(sep-aug) DESC;
SQL
cat > output/channel.sql <<'SQL'
WITH base AS (
 SELECT *, (ROUND(CAST(line_paid_amount AS DECIMAL(18,2))*CAST(commission_rate AS DECIMAL(6,4)),2)-ROUND(COALESCE(CAST(refund_amount AS DECIMAL(18,2)),0)*CAST(commission_rate AS DECIMAL(6,4)),2)) AS net_line,
 strftime(CAST(completed_at AS DATE), '%Y-%m') AS month
 FROM data WHERE order_kind='customer_sale' AND status='completed' AND CAST(completed_at AS DATE)>=DATE '2026-08-01' AND CAST(completed_at AS DATE)<DATE '2026-10-01'
), g AS (SELECT channel, SUM(CASE WHEN month='2026-08' THEN net_line ELSE 0 END) aug, SUM(CASE WHEN month='2026-09' THEN net_line ELSE 0 END) sep FROM base GROUP BY channel)
SELECT channel, aug, sep, sep-aug AS change FROM g ORDER BY ABS(sep-aug) DESC;
SQL
